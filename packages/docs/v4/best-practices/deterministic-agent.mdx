**Planned. Not available in Stagehand v4.**

The original guide turned nondeterministic agent runs into replayable scripts using a local cache directory, then analyzed them with the history API. Stagehand v4 has no agent, no local file cache, and no history accessor.
</Warning>

## What to use instead

v4 gets you determinism from the other direction: rather than recording an agent's exploration, you write the steps explicitly and let [server-side caching](/v4/best-practices/caching) strip the inference out of repeat runs. Once an action is cached, the run is a deterministic replay of the recorded selectors with no LLM call at all.

<View title="TypeScript" icon="js">
```typescript
const stagehand = new Stagehand({
  apiKey: process.env.BROWSERBASE_API_KEY,
  browser: { type: "browserbase" },
  cache: { threshold: 1 }, // Replay after a single successful run
});

await stagehand.init();
const page = await stagehand.context.activePage();
await page.goto("https://example.com/login");

// v4 reads no environment variables, so read the credentials yourself
const email = process.env.LOGIN_EMAIL;
const password = process.env.LOGIN_PASSWORD;

// First run: inference. Every run after: replayed from the cache.
await stagehand.act("type %email% into the Email address field", {
  variables: { email },
});
await stagehand.act("type %password% into the Password field", {
  variables: { password },
});
await stagehand.act("click the Sign in button");
```
</View>

<View title="Python" icon="python">
```python
from stagehand import CacheOptions, Stagehand

stagehand = Stagehand(
    api_key=os.environ["BROWSERBASE_API_KEY"],
    browser="browserbase",
    cache=CacheOptions(threshold=1),  # Replay after a single successful run
)

await stagehand.init()
page = await stagehand.context.active_page()
await page.goto("https://example.com/login")

# v4 reads no environment variables, so read the credentials yourself
email = os.environ["LOGIN_EMAIL"]
password = os.environ["LOGIN_PASSWORD"]

# First run: inference. Every run after: replayed from the cache.
await stagehand.act(
    "type %email% into the Email address field",
    variables={"email": email},
)
await stagehand.act(
    "type %password% into the Password field",
    variables={"password": password},
)
await stagehand.act("click the Sign in button")
```
</View>

<View title="Go" icon="golang">
```go
apiKey := os.Getenv("BROWSERBASE_API_KEY")
cache := stagehand.CacheWithThreshold(1) // Replay after a single successful run

client := stagehand.New(stagehand.StagehandClientInitParams{
	APIKey:  &apiKey,
	Browser: stagehand.BrowserbaseClientBrowserSource{},
	Cache:   &cache,
})

if err := client.Init(ctx); err != nil {
	return err
}

browserContext, err := client.Context()
if err != nil {
	return err
}
page, err := browserContext.ActivePage(ctx)
if err != nil {
	return err
}
if err := page.Goto(ctx, "https://example.com/login", nil); err != nil {
	return err
}

// v4 reads no environment variables, so read the credentials yourself
email := os.Getenv("LOGIN_EMAIL")
password := os.Getenv("LOGIN_PASSWORD")

// First run: inference. Every run after: replayed from the cache.
if _, err := client.Act(ctx, "type %email% into the Email address field", &stagehand.StagehandClientActOptions{
	ActOptions: stagehand.ActOptions{
		Variables: stagehand.Variables{
			"email": stagehand.PrimitiveVariable(stagehand.StringVariable(email)),
		},
	},
}); err != nil {
	return err
}

if _, err := client.Act(ctx, "type %password% into the Password field", &stagehand.StagehandClientActOptions{
	ActOptions: stagehand.ActOptions{
		Variables: stagehand.Variables{
			"password": stagehand.PrimitiveVariable(stagehand.StringVariable(password)),
		},
	},
}); err != nil {
	return err
}

_, err = client.Act(ctx, "click the Sign in button", nil)
```
</View>

<Note>
Cached actions replay with self-healing turned off, so a run either reproduces exactly or falls back to fresh inference. That fallback is the invalidation story: when the site changes, the next run re-learns the step and re-primes the cache.
</Note>

For a fully deterministic path with no model in the loop at all, plan once with `observe()` and then drive the recorded selectors through locators. See [Plan then execute](/v4/basics/observe#plan-then-execute).
