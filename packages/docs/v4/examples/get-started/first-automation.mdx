**Goal:** Learn Stagehand's three AI primitives by automating [stagehand.dev](https://stagehand.dev) in one script.

**Outcome:** A working TypeScript, Python, or Go script that extracts the value proposition, clicks through to Evals, and lists clickable actions — ready to adapt to your own site.

## Agent brief

Give this section to your coding agent as the implementation brief. It should adapt the target site and schema rather than scaffold a separate app.

```text
Build a first Stagehand v4 automation script.

Stack: @browserbasehq/stagehand (or stagehand Python / Go SDK), zod.
Browser: Browserbase via BROWSERBASE_API_KEY (cloud) — no local Chromium
required. Model: omit on Browserbase so Model Gateway selects one.

Flow:
1. Launch a cloud browser and attach Stagehand with Stagehand.create.
2. goto https://stagehand.dev.
3. extract() the value proposition using a strict schema
   ({ valueProposition: string }).
4. act("Click the 'Evals' button.").
5. observe("What can I click on this page?") and log candidates.
6. Close Stagehand AND the browser in finally/defer blocks.

Rules:
- Never hard-code API keys; read from environment only.
- Prefer locators for stable selectors; use AI primitives only where
  judgment is needed.
- Every primitive call must handle failure without leaking the session.

Verify: script prints the extracted value proposition, then a list of
observed actions, and exits 0 with no leaked sessions.
```

  New to Stagehand? Install and run the shorter [Quickstart](/v4/first-steps/quickstart) under Docs first, then come back here for the cookbook-shaped walkthrough.

## Stack

- Node.js 18+, Python 3.10+, or Go 1.22+
- `@browserbasehq/stagehand` / `stagehand` / SDK Go module
- `BROWSERBASE_API_KEY` for a cloud browser
- Optional model via [Model Gateway](/v4/configuration/models) (omit `model` on Browserbase)

<Tab title="TypeScript">
```typescript
import { browserbase, Stagehand } from "@browserbasehq/stagehand";
import { z } from "zod/v4";

async function main() {
  const browser = await browserbase.launch({
    apiKey: process.env.BROWSERBASE_API_KEY,
  });

  try {
    const stagehand = await Stagehand.create({ browser });

    try {
      const [page] = await browser.context.pages();
      await page.goto("https://stagehand.dev");

      // Extract: pull structured data with a schema
      const { data: pitch } = await stagehand.extract(
        "Extract the value proposition from the page.",
        z.object({ valueProposition: z.string() }),
      );
      console.log("Value proposition:", pitch.valueProposition);

      // Act: natural-language click
      await stagehand.act("Click the 'Evals' button.");

      // Observe: discover what you can do next
      const { data: actions } = await stagehand.observe(
        "What can I click on this page?",
      );
      console.log("Observed actions:", actions);
    } finally {
      await stagehand.close();
    }
  } finally {
    await browser.close();
  }
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
```

```python
import asyncio
import os

from pydantic import BaseModel
from stagehand import Stagehand, browserbase


class ValueProposition(BaseModel):
    value_proposition: str


async def main() -> None:
    browser = await browserbase.launch(
        api_key=os.environ["BROWSERBASE_API_KEY"],
    )
    try:
        stagehand = await Stagehand.create(browser=browser)
        try:
            page = (await browser.context.pages())[0]
            await page.goto("https://stagehand.dev")

            pitch = await stagehand.extract(
                "Extract the value proposition from the page.",
                ValueProposition,
            )
            print("Value proposition:", pitch.data.value_proposition)

            await stagehand.act("Click the 'Evals' button.")

            actions = await stagehand.observe("What can I click on this page?")
            print("Observed actions:", actions.data)
        finally:
            await stagehand.close()
    finally:
        await browser.close()


asyncio.run(main())
```

<Tab title="Go">
```go
package main

import (
	"context"
	"errors"
	"fmt"
	"log"
	"os"

	stagehand "github.com/browserbase/stagehand/packages/sdk-go/v4"
)

type valueProposition struct {
	ValueProposition string `json:"value_proposition"`
}

func main() {
	if err := run(context.Background()); err != nil {
		log.Fatal(err)
	}
}

func run(ctx context.Context) (err error) {
	apiKey := os.Getenv("BROWSERBASE_API_KEY")

	browser, err := stagehand.LaunchBrowserbase(ctx, stagehand.BrowserbaseLaunchOptions{APIKey: apiKey})
	if err != nil {
		return err
	}
	defer func() { err = errors.Join(err, browser.Close(ctx)) }()

	client, err := stagehand.Create(ctx, stagehand.CreateOptions{Browser: browser})
	if err != nil {
		return err
	}
	defer func() { err = errors.Join(err, client.Close(ctx)) }()

	browserContext, err := browser.Context()
	if err != nil {
		return err
	}
	pages, err := browserContext.Pages(ctx)
	if err != nil {
		return err
	}
	if len(pages) == 0 {
		return errors.New("Stagehand initialized without an active page")
	}
	page := pages[0]

	if _, err := page.Goto(ctx, "https://stagehand.dev", nil); err != nil {
		return err
	}

	extracted, err := stagehand.Extract[valueProposition](
		ctx,
		client,
		"Extract the value proposition from the page.",
		nil,
	)
	if err != nil {
		return err
	}
	fmt.Println("Value proposition:", extracted.Data.ValueProposition)

	if _, err := client.Act(ctx, stagehand.ActInstruction("Click the 'Evals' button."), nil); err != nil {
		return err
	}

	instruction := "What can I click on this page?"
	observeResult, err := client.Observe(ctx, &instruction, nil)
	if err != nil {
		return err
	}
	fmt.Println("Observed actions:", observeResult.Data)

	return nil
}
```

## Walkthrough

1. **Launch a browser** — `browserbase.launch` (or the Go equivalent) creates a cloud browser. Pass your API key from the environment; never hard-code it.
2. **Create Stagehand** — `Stagehand.create({ browser })` attaches the AI primitives to that browser. Model Gateway picks a model when you omit `model` on Browserbase.
3. **`goto` then `extract`** — Navigate with Playwright-style APIs, then extract structured fields with a schema (`zod` / Pydantic / Go struct).
4. **`act` then `observe`** — Use natural language for clicks and discovery. Prefer locators when selectors are stable; reserve AI for steps that need judgment.
5. **Close both handles** — Always close Stagehand and the browser in `finally` / `defer` so sessions do not leak.

## Check your work

- The script prints the extracted value proposition before any click happens.
- `act` runs against the Evals button; `observe` returns a non-empty action list on the Evals page.
- Both close calls run even when a primitive throws — check the `finally` / `defer` blocks.
- No API key appears in source or logs.
- Swapping the target URL and schema is the only change needed for your own site.
